gdAlertify JS

Instalación

<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/FelipeGaos/gdAlertify@v1.0.0/dist/gdAlertify.min.css">
<script src="https://cdn.jsdelivr.net/gh/FelipeGaos/gdAlertify@v1.0.0/dist/gdAlertify.min.js"></script>

Ejemplos

gdAlertify.alertSuccess('Operación realizada con éxito');
                
gdAlertify.alertError('Ha ocurrido un error inesperado');
                
gdAlertify.confirm({
    title: 'Confirmar acción',
    message: '¿Está seguro de que desea continuar?',
    onConfirm: () => { gdAlertify.notify('Acción confirmada', 'success'); },
});
                
gdAlertify.notify('Notificación de advertencia', 'warning');
                
gdAlertify.showLoader({
    title: 'Cargando, por favor espere...',
    type: 'spinner',
    color: '#fff'
});
gdAlertify.hideLoader();
                
<- Atajos para Alertas ->
gdAlertify.alertSuccess('¡Éxito! Operación realizada correctamente.');
gdAlertify.alertError('¡Error! Algo salió mal.');
gdAlertify.alertInfo('Este es un mensaje informativo.');
gdAlertify.alertWarning('Advertencia: Acción no recomendada.');

<- Atajos para Notificaciones ->
gdAlertify.notiSuccess('Operación exitosa, notificación.');
gdAlertify.notiError('Error al procesar la solicitud, notificación.');
gdAlertify.notiInfo('Información importante, notificación.');
gdAlertify.notiWarning('Advertencia en proceso, notificación.');

<- Atajos para Confirmaciones ->
gdAlertify.confirm({
    title: 'Confirmar acción',
    message: '¿Está seguro de que desea continuar?',
    onConfirm: () => { console.log('Acción confirmada'); }
});

<- Atajos para Loaders ->
gdAlertify.showLoader({
    title: 'Cargando, por favor espere...',
    type: 'spinner',
    color: '#ff0000'
});
gdAlertify.hideLoader();
                
<- Propiedades de Configuración ->
gdAlertifyConfig({
    backdropColor: 'rgba(0, 0, 0, 0.5)',  // Color de fondo del modal
    modalColor: '#fff',                   // Color de fondo del modal
    modalTextColor: '#000',               // Color del texto en el modal
    defaultButtonClass: 'btn-primary',    // Clase por defecto para los botones
    confirmButtonClass: 'btn-success',    // Clase del botón de confirmar
    cancelButtonClass: 'btn-danger',      // Clase del botón de cancelar
    defaultConfirmText: 'Confirmar',      // Texto por defecto del botón de confirmar
    defaultCancelText: 'Cancelar',        // Texto por defecto del botón de cancelar
    defaultOkText: 'Aceptar',             // Texto por defecto del botón de alerta
    canCloseOutside: true,                // Permite cerrar el modal haciendo clic fuera de él
    theme: 'light',                       // Tema por defecto de la alerta/modal
    defaultNotificationPosition: 'top-right', // Posición por defecto de las notificaciones
    loaderDefaults: {                     // Configuración del loader
        title: 'Cargando...',            // Texto para el loader
        type: 'spinner',                 // Tipo (Actualmente tenemos 1 `spinner`)
        color: '#000',                   // Color en HEX para el texto
        spinnerColor: '#ff0000'          // Color en HEX para el spinner
    }
});
